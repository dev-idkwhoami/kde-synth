cmake_minimum_required(VERSION 3.21)
project(kde_synth VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_INSTALL_DOCDIR "share/doc/kde-synth" CACHE PATH "Documentation directory")
include(GNUInstallDirs)
add_compile_definitions(KDE_SYNTH_AUDIO_PATH="${CMAKE_INSTALL_FULL_LIBEXECDIR}/kde-synth/kde-synth-audio")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick Widgets DBus ShaderTools)
find_package(LayerShellQt REQUIRED)
set(CONTROLLER_SOURCES src/Controller.cpp src/Controller.h src/ActivityMonitor.cpp src/ActivityMonitor.h src/DesktopLayout.cpp src/DesktopLayout.h)
qt_add_executable(kde-synth src/main.cpp ${CONTROLLER_SOURCES} src/SettingsWindow.cpp src/Wallpaper.cpp src/Wallpaper.h)
qt_add_resources(kde-synth visuals PREFIX "/" FILES src/Bars.qml src/Trace.qml src/desktop-layout.js)
qt_add_shaders(kde-synth effects PREFIX "/" FILES src/wallpaper.frag)
target_link_libraries(kde-synth PRIVATE Qt6::Quick Qt6::Widgets Qt6::DBus LayerShellQt::Interface)
find_program(GO_EXECUTABLE go REQUIRED)
add_custom_command(OUTPUT "${CMAKE_SOURCE_DIR}/bin/kde-synth-audio"
    COMMAND ${CMAKE_COMMAND} -E env "GOCACHE=${CMAKE_BINARY_DIR}/go-cache"
        ${GO_EXECUTABLE} build -trimpath -buildvcs=false -o "${CMAKE_SOURCE_DIR}/bin/kde-synth-audio" ./audio
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    DEPENDS audio/main.go audio/spectrum.go audio/routing.go audio/waveform.go go.mod)
add_custom_target(audio-helper ALL DEPENDS "${CMAKE_SOURCE_DIR}/bin/kde-synth-audio")
add_dependencies(kde-synth audio-helper)
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(kde-synth-tests tests/controller_test.cpp ${CONTROLLER_SOURCES})
    target_include_directories(kde-synth-tests PRIVATE src)
    target_link_libraries(kde-synth-tests PRIVATE Qt6::Core Qt6::DBus)
    add_test(NAME settings-and-envelope COMMAND kde-synth-tests)
endif()

install(TARGETS kde-synth RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS "${CMAKE_SOURCE_DIR}/bin/kde-synth-audio"
    DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/kde-synth)
install(FILES packaging/dev.kdesynth.Visualizer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/logo.png DESTINATION ${CMAKE_INSTALL_DATADIR}/pixmaps RENAME kde-synth.png)
install(FILES README.md THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
if(EXISTS "${CMAKE_SOURCE_DIR}/LICENSE")
    install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/kde-synth)
endif()
install(DIRECTORY docs DESTINATION ${CMAKE_INSTALL_DOCDIR})
